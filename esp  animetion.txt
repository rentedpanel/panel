static pid_t find_pid(const char* proc_name) {
DIR* dir = opendir("/proc");
if (!dir) return -1;
struct dirent* entry;
while ((entry = readdir(dir))) {

entry->d_name);

if (fgets(line, sizeof(line), f)) {
line[strcspn(line, "\n")] = 0;

closedir(dir);

}

fclose(f);
}
closedir(dir);
return -1;
}

static bool world_to_screen(const Vec3& pos, const
Mat4& vp,
float sw, float sh,
float& sx, float& sy) {
float clipX = vp.m[0]*pos.x + vp.m[4]*pos.y +
vp.m[8]*pos.z

+ vp.m[12];

float clipY = vp.m[1]*pos.x + vp.m[5]*pos.y +
vp.m[9]*pos.z

+ vp.m[13];

float clipW = vp.m[3]*pos.x + vp.m[7]*pos.y +
vp.m[11]*pos.z + vp.m[15];
if (clipW < 0.001f) return false;
float ndcX = clipX / clipW;
float ndcY = clipY / clipW;
sx = (ndcX + 1.0f) * 0.5f * sw;
sy = (1.0f - ndcY) * 0.5f * sh;
return true;
}

resultCode: Int, data: Intent?) {
